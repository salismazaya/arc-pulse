3. Solidity Integration

Consume rates directly in contracts targeting the Osaka/Cancun EVM baseline.

Interface Definition 
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

interface IArcPulseOracle {
    function latestRoundData(bytes32 feedId) external view returns (
        uint80 roundId,
        int256 answer,
        uint256 startedAt,
        uint256 updatedAt,
        uint80 answeredInRound
    );

    function getLatestPrice(bytes32 feedId) external view returns (
        int256 price,
        uint8 decimals
    );
}
Consumer Contract Example 
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

import "./IArcPulseOracle.sol";

contract MerchantCheckout {
    IArcPulseOracle public immutable oracle;
    bytes32 public constant IDR_FEED = 0x27be29a97b92db0e31ca03433d2ea3070a6b370258fa5d1f0c302514746955a9;

    constructor(address _oracle) {
        oracle = IArcPulseOracle(_oracle);
    }

    /// Calculate required native Arc USDC (18 decimals) for an IDR invoice
    function calculateUsdcRequired(uint256 idrAmount) public view returns (uint256 usdcWei) {
        (int256 rate, uint8 decimals) = oracle.getLatestPrice(IDR_FEED);
        require(rate > 0, "Invalid rate");

        // rate has 8 decimals (e.g. 1585000000000 = 15,850 IDR/USD)
        // native USDC has 18 decimals
        usdcWei = (idrAmount * (10 ** decimals) * 1e18) / uint256(rate);
    }
}

4. Frontend Integration (Ethers & Viem)

Querying feeds from web apps or Node/Bun services.

Ethers.js (v6) 
import { ethers } from "ethers";

const RPC_URL = "https://rpc.mainnet.arc.io";
const ORACLE_ADDR = "0x0E5A4f7C4CDAA2C1cc240EeF69f9eedD925Fafc8";
const IDR_FEED = "0x27be29a97b92db0e31ca03433d2ea3070a6b370258fa5d1f0c302514746955a9";

const ABI = [
  "function latestRoundData(bytes32 feedId) external view returns (uint80 roundId, int256 answer, uint256 startedAt, uint256 updatedAt, uint80 answeredInRound)",
  "function getLatestPrice(bytes32 feedId) external view returns (int256 price, uint8 decimals)"
];

async function checkPrice() {
  const provider = new ethers.JsonRpcProvider(RPC_URL);
  const oracle = new ethers.Contract(ORACLE_ADDR, ABI, provider);

  const [price, decimals] = await oracle.getLatestPrice(IDR_FEED);
  const formattedRate = Number(price) / (10 ** Number(decimals));

  console.log("Current USD/IDR Rate:", formattedRate);
}

checkPrice();

});

console.log("Raw Rate:", price, "Decimals:", decimals);

5. Arc Native Economics & Math

Understanding dual decimals and gas parameters on Arc L1.

Dual Decimals System

Arc uses USDC as its native currency. Native transfers (msg.value) use 18 decimals (1 USDC = 1018 wei). The ERC-20 interface (0x3600000000000000000000000000000000000000) uses 6 decimals (1 USDC = 106 units). Never mix the two without a 1012 conversion factor.

Settlement Formula for Native USDC 
Native_USDC_Wei = (Fiat_Amount * 10^decimals * 10^18) / Rate
Mempool Minimum Gas Floor

Always pass gasPrice: 20000000000n (20 Gwei) when broadcasting contract writes. Transactions below 20 Gwei will be dropped silently without being mined.
